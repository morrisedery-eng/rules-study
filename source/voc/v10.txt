P1| שֶׁטַח פְּנִימִי: 35 רֶגֶל רָבוּעַ שֶׁטַח שָׁמִישׁ לְכׇל יֶלֶד (20 רַק לְמִתְקָנִים בְּרִישָּׁיוֹן רָצִיף מֵאָז 1 בְּאוֹקְטוֹבֶּר 1992); כׇּל כִּיתָּה צְרִיכָה 35. מִטְבָּח, מִשְׂרָדִים, מִסְדְּרוֹנוֹת, שֵׁירוּתִים וּמַחְסָנִים לֹא נִסְפָּרִים. הַתְּפוּסָה הַמּוּרְשֵׁית נִקְבַּעַת לְפִי הַגּוֹרֵם הַמַּגְבִּיל בְּיוֹתֵר מִבֵּין שֶׁטַח פְּנִימִי, חָצֵר, בִּיּוּב וְשֵׁירוּתִים, וּמְפוּרְסֶמֶת בְּכׇל חֶדֶר.
E1| כִּיתָּה שֶׁל 525 רֶגֶל רָבוּעַ: 525 ÷ 35 = 15 יְלָדִים.
I1| 35 בִּפְנִים. הַמִּסְפָּר הַקָּטָן קוֹבֵעַ.
P2| שֶׁטַח חִיצוֹנִי: 45 רֶגֶל רָבוּעַ לְכׇל יֶלֶד מִגִּיל שָׁנָה, מַסְפִּיק לְפָחוֹת לְמַחְצִית הַתְּפוּסָה. גָּדֵר בְּגוֹבַהּ 4 רֶגֶל לְפָחוֹת, בְּלִי רְוָוחִים מֵעַל 3.5 אִינְץ׳, לְפָחוֹת שְׁתֵּי יְצִיאוֹת (אַחַת רְחוֹקָה מֵהַבִּנְיָין), חוֹבָה צֵל, וּמִפְגְּעֵי מַיִם מְגוּדָּרִים.
E2| רִישָּׁיוֹן לְ-60: הֶחָצֵר חַיֶּיבֶת לְהָכִיל לְפָחוֹת 30 × 45 = 1,350 רֶגֶל רָבוּעַ.
I2| 45 בַּחוּץ, לְפָחוֹת לַחֲצִי.
P3| שֵׁירוּתִים: אַסְלָה וְכִיּוֹר לְ-15 הַיְּלָדִים הָרִאשׁוֹנִים, וְעוֹד אֶחָד מִכׇּל סוּג לְכׇל 30 נוֹסָפִים (רַק תִּינוֹקוֹת בְּחִיתּוּלִים: אַסְלָה וְ-2 כִּיּוֹרִים לְכׇל 30). שֵׁירוּתִים לֹא נִפְתָּחִים לְאֵזוֹר הֲכָנַת מָזוֹן. אַסְלוֹת וְכִיּוֹרִים מְנוּקִּים וּמְחוּטָּאִים לְפָחוֹת פַּעַם בַּיּוֹם.
E3| 75 יְלָדִים: 1 + (60 ÷ 30 = 2) = 3 אַסְלוֹת וְ-3 כִּיּוֹרִים.
I3| 15 הָרִאשׁוֹנִים, אַחַר כָּךְ כׇּל 30.
P4| טֶמְפֶּרָטוּרָה 65–82°F; תְּאוּרָה לְפָחוֹת 20 פוּט-קַנְדֶּל (50 לַעֲבוֹדָה מִקָּרוֹב); רְשָׁתוֹת בְּחַלּוֹנוֹת פְּתוּחִים. בְּדִיקָה יוֹמִית שֶׁל אֵזוֹרֵי מִשְׂחָק בִּפְנִים וּבַחוּץ; יוֹמַן הַבְּדִיקָה הַפְּנִימִית נִשְׁמָר 12 חוֹדָשִׁים. שְׁקָעִים מוּגָנִים אוֹ כִּיסּוּיִים; מַעֲקֶה כְּשֶׁהַנְּפִילָה מֵעַל 30 אִינְץ׳; חֲבָלִים מְסוּכָּנִים מִחוּץ לְהֶישֵּׂג יָד; גַּלַּאי פַּחְמָן חַד-חַמְצָנִי כְּשֶׁנִּדְרָשׁ.
E4| אִם הַמַּזְגָן מִתְקַלְקֵל וְהַחֶדֶר מַגִּיעַ לְ-85°F, זוֹ הֲפָרָה.
I4| 65–82, כְּמוֹ בַּיִת נוֹחַ.
P5| חוֹמָרִים מְסוּכָּנִים (נִיקּוּי, כִימִיקָלִים, סַכִּינִים, שַׂקִּיּוֹת, מַצִּיתִים) מְסוּמָּנִים וּנְעוּלִים אוֹ מִחוּץ לְהֶישֵּׂג יָד. נֶשֶׁק אָסוּר (לְמַעֵט אֲכִיפַת הַחוֹק וּקְצִינֵי בִּיטָּחוֹן בֵּית סֵפֶר). אָסוּר לְעַשֵּׁן אוֹ לְאַדּוֹת בְּשׁוּם מָקוֹם שֶׁיֵּשׁ יְלָדִים, כּוֹלֵל רְכָבִים וְטִיּוּלִים; הַהוֹרִים מְקַבְּלִים הוֹדָעָה בִּכְתָב. בַּעֲלֵי חַיִּים מְחוּסָּנִים (אֶלָּא אִם וֶטֶרִינָר מְתַעֵד אַחֶרֶת), בִּכְלוּב בַּכִּיתָּה, לְעוֹלָם לֹא לְיַד מָזוֹן; הַהוֹרִים מְקַבְּלִים הוֹדָעָה בִּכְתָב.
E5| הָאוֹגֵר שֶׁל הַכִּיתָּה גָּר בִּכְלוּב נָקִי, וְהַהוֹרִים קִיבְּלוּ מִכְתָּב שֶׁיֵּשׁ חַיַּית מַחְמָד בַּחֶדֶר.
I5| בְּלִי נֶשֶׁק, בְּלִי עִישּׁוּן, כִימִיקָלִים נְעוּלִים.
P6| קְצִינֵי בִּיטָּחוֹן בֵּית סֵפֶר (שׁוֹטֵר, קְצִין בִּיטָּחוֹן, שׁוֹמֵר) הֵם רְשׁוּת. הֵם לֹא נִסְפָּרִים בַּיַּחַס וְלֹא מַשְׁגִּיחִים יְשִׁירוֹת עַל יְלָדִים, הַמִּשְׁפָּחוֹת מְקַבְּלוֹת הוֹדָעָה בִּכְתָב, וְהַשֶּׁרִיף וּמִשְׂרַד בָּתֵּי סֵפֶר בְּטוּחִים מְקַבְּלִים הוֹדָעָה תּוֹךְ 72 שָׁעוֹת אִם קָצִין פּוּטַר בִּגְלַל הִתְנַהֲגוּת אוֹ יָרָה מִחוּץ לָאִימּוּן.
E6| מֶרְכָּז שׂוֹכֵר שׁוֹמֵר. הוּא עוֹמֵד בַּדֶּלֶת אֲבָל לֹא נִסְפָּר כְּמוֹרֶה.

P7| מִתְקְנֵי חָצֵר: מְעוּגָּנִים, נִבְדָּקִים כׇּל חוֹדֶשׁ (רִישּׁוּם 12 חוֹדָשִׁים). מִתְקָן קָבוּעַ צָרִיךְ אֵזוֹר נְפִילָה שֶׁל 6 רֶגֶל לְכׇל כִּיוּוּן; כִּיסּוּי רוֹפֵף (מַאלְץ׳, גּוּמִי, חוֹל) בְּעוֹמֶק 6 אִינְץ׳ לְפָחוֹת; לְעוֹלָם לֹא אַסְפַלְט, בֵּטוֹן, עָפָר אוֹ דֶּשֶׁא מִתַּחַת לְמִתְקְנֵי טִיפּוּס. אַרְגַּז חוֹל מְכוּסֶּה בְּכׇל לַיְלָה. בְּרֵיכוֹת פָּעוֹטוֹת אֲסוּרוֹת. בִּפְנִים: רָהִיטִים כְּבֵדִים מְקוּבָּעִים; לוּחַ נִיקָּיוֹן כָּתוּב; צַעֲצוּעַ עִם רוֹק מְחוּטָּא לִפְנֵי יֶלֶד אַחֵר.
